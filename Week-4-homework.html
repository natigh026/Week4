gid_selected <- gid %>%
  select(iso3, country, gii_2010, gii_2019) |> 
  mutate(
    gii_2010 = as.numeric(gii_2010),
    gii_2019 = as.numeric(gii_2019)
  ) |> 
  filter(!is.na(iso3))

  mutate(

  )

  mutate(
